cmake_minimum_required(VERSION 3.20)
project(aisthra_installed_consumer LANGUAGES C)
find_package(aisthra CONFIG REQUIRED)
option(AIS_USE_SHARED "Use shared Aisthra target" OFF)
add_executable(aisthra_installed_consumer consumer.c)
if(AIS_USE_SHARED)
    target_link_libraries(aisthra_installed_consumer PRIVATE aisthra::shared)
else()
    target_link_libraries(aisthra_installed_consumer PRIVATE aisthra::static)
endif()
